<script>
(function(){
  var state = {
    gender: null,
    skin: '#f2c9a0',
    hair: '#7a4a2a',
    outfit: '#3d5a8a',
    cap: '#d94b3f',
    capOn: true
  };

  var hairOptions = ['#7a4a2a','#2b2b2b','#c98a3a','#8a4b8a','#d9483f'];
  var outfitOptions = ['#3d5a8a','#d94b3f','#3a8a5a','#8a3a7a','#e0a63a'];
  var capOptions = ['#d94b3f','#3d5a8a','#3a8a5a','#f2c94c','none'];

  function buildSwatches(container, options, key){
    container.innerHTML = '';
    options.forEach(function(c){
      var el = document.createElement('div');
      el.className = 'swatch';
      el.style.background = c === 'none' ? 'repeating-conic-gradient(#ccc 0% 25%, #eee 0% 50%) 50% / 10px 10px' : c;
      if ((key==='cap' && ((c==='none' && !state.capOn) || (c===state.cap && state.capOn))) ||
          (key!=='cap' && state[key]===c)) el.classList.add('active');
      el.addEventListener('click', function(){
        if (key === 'cap') {
          if (c === 'none') { state.capOn = false; } else { state.capOn = true; state.cap = c; }
        } else {
          state[key] = c;
        }
        buildSwatches(container, options, key);
        render();
      });
      container.appendChild(el);
    });
  }

  function render(){
    var svg = document.getElementById('charSvg');
    var skirt = state.gender === 'female';
    svg.innerHTML =
      '<rect x="0" y="0" width="300" height="190" fill="#8fd4e8"/>' +
      '<rect x="0" y="190" width="300" height="110" fill="#e8d29a"/>' +
      '<ellipse cx="230" cy="200" rx="60" ry="16" fill="#4f8a52" opacity="0.5"/>' +
      '<rect x="255" y="70" width="14" height="140" fill="#7a5a3a"/>' +
      '<ellipse cx="262" cy="70" rx="45" ry="30" fill="#3f8a4a"/>' +
      '<circle cx="60" cy="150" r="24" fill="#ffffff" opacity="0.85"/>' +
      '<circle cx="85" cy="150" r="18" fill="#ffffff" opacity="0.85"/>' +
      // body/outfit
      '<rect x="120" y="185" width="60" height="' + (skirt ? '55' : '65') + '" rx="12" fill="' + state.outfit + '"/>' +
      (skirt ? '<path d="M115 225 L185 225 L195 260 L105 260 Z" fill="' + state.outfit + '"/>' : '<rect x="120" y="245" width="24" height="35" fill="#3a3a3a"/><rect x="156" y="245" width="24" height="35" fill="#3a3a3a"/>') +
      // arms
      '<rect x="102" y="190" width="18" height="45" rx="8" fill="' + state.skin + '"/>' +
      '<rect x="180" y="190" width="18" height="45" rx="8" fill="' + state.skin + '"/>' +
      // head
      '<circle cx="150" cy="150" r="42" fill="' + state.skin + '"/>' +
      // hair back
      (skirt ? '<path d="M108 140 Q100 200 120 210 Q112 150 130 118 Z" fill="' + state.hair + '"/><path d="M192 140 Q200 200 180 210 Q188 150 170 118 Z" fill="' + state.hair + '"/>' : '') +
      // hair top
      '<path d="M108 138 Q108 100 150 96 Q192 100 192 138 Q170 118 150 118 Q130 118 108 138 Z" fill="' + state.hair + '"/>' +
      // face
      '<circle cx="136" cy="152" r="4" fill="#3a2a1a"/>' +
      '<circle cx="164" cy="152" r="4" fill="#3a2a1a"/>' +
      '<path d="M138 168 Q150 176 162 168" stroke="#7a4a35" stroke-width="3" fill="none" stroke-linecap="round"/>' +
      '<circle cx="122" cy="162" r="6" fill="#e69a8a" opacity="0.5"/>' +
      '<circle cx="178" cy="162" r="6" fill="#e69a8a" opacity="0.5"/>' +
      // cap
      (state.capOn ? '<path d="M104 132 Q150 90 196 132 Q196 118 150 108 Q104 118 104 132 Z" fill="' + state.cap + '"/><rect x="150" y="106" width="46" height="14" rx="6" fill="' + state.cap + '"/>' : '');
  }

  function updateGenderUI(){
    document.getElementById('femaleBtn').classList.toggle('selected', state.gender==='female');
    document.getElementById('maleBtn').classList.toggle('selected', state.gender==='male');
  }

  document.getElementById('femaleBtn').addEventListener('click', function(){
    state.gender = 'female';
    document.getElementById('genderHint').textContent = '';
    updateGenderUI(); render();
  });
  document.getElementById('maleBtn').addEventListener('click', function(){
    state.gender = 'male';
    document.getElementById('genderHint').textContent = '';
    updateGenderUI(); render();
  });

  document.getElementById('randomBtn').addEventListener('click', function(){
    state.skin = ['#f2c9a0','#e0a878','#c98858','#8a5a3a'][Math.floor(Math.random()*4)];
    state.hair = hairOptions[Math.floor(Math.random()*hairOptions.length)];
    state.outfit = outfitOptions[Math.floor(Math.random()*outfitOptions.length)];
    var capChoice = capOptions[Math.floor(Math.random()*capOptions.length)];
    if (capChoice === 'none') { state.capOn = false; } else { state.capOn = true; state.cap = capChoice; }
    if (!state.gender) { state.gender = Math.random() < 0.5 ? 'female' : 'male'; updateGenderUI(); }
    render();
    showToast('New look generated');
  });

  document.getElementById('customizeBtn').addEventListener('click', function(){
    buildSwatches(document.getElementById('hairSwatches'), hairOptions, 'hair');
    buildSwatches(document.getElementById('outfitSwatches'), outfitOptions, 'outfit');
    buildSwatches(document.getElementById('capSwatches'), capOptions, 'cap');
    document.getElementById('panelOverlay').classList.add('open');
  });
  document.getElementById('cancelPanel').addEventListener('click', function(){
    document.getElementById('panelOverlay').classList.remove('open');
  });
  document.getElementById('savePanel').addEventListener('click', function(){
    document.getElementById('panelOverlay').classList.remove('open');
    showToast('Look saved');
  });

  var nameInput = document.getElementById('nameInput');
  nameInput.addEventListener('input', function(){
    var v = nameInput.value.trim();
    var err = document.getElementById('nameError');
    if (!v) { err.textContent = 'Enter a name to continue'; }
    else if (v.length < 2) { err.textContent = 'Name must be at least 2 characters'; }
    else { err.textContent = ''; }
  });

  function showToast(msg){
    var t = document.getElementById('toast');
    t.textContent = msg;
    t.classList.add('show');
    setTimeout(function(){ t.classList.remove('show'); }, 1400);
  }

  document.getElementById('playBtn').addEventListener('click', function(){
    var name = nameInput.value.trim();
    var err = document.getElementById('nameError');
    var genderHint = document.getElementById('genderHint');
    var ok = true;
    if (!name || name.length < 2) { err.textContent = 'Enter a name to continue'; ok = false; }
    if (!state.gender) { genderHint.textContent = 'Choose a gender to continue'; ok = false; }
    if (!ok) return;
    showToast('Welcome, ' + name + '! Loading game…');
  });

  render();
})();
</script>
