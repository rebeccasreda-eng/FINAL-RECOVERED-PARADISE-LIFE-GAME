<script>
(function(){
  var GRID = 11;
  var TW = 64, TH = 32;
  var world = document.getElementById('world');
  var viewport = document.getElementById('viewport');

  // 0 = water, 1 = sand, 2 = grass
  var map = [];
  var cx = (GRID-1)/2, cy = (GRID-1)/2;
  for (var y = 0; y < GRID; y++){
    map[y] = [];
    for (var x = 0; x < GRID; x++){
      var d = Math.hypot(x-cx, y-cy);
      map[y][x] = d < 3.1 ? 2 : (d < 4.2 ? 1 : 0);
    }
  }

  function isoX(x, y){ return (x - y) * (TW/2); }
  function isoY(x, y){ return (x + y) * (TH/2); }

  function tileColor(type){
    return type === 2 ? '#5aab3e' : type === 1 ? '#e8d29a' : '#3f8fc9';
  }
  function tileColorDark(type){
    return type === 2 ? '#3f8a2a' : type === 1 ? '#cdb277' : '#2f70a3';
  }

  function tileSvg(type){
    var top = tileColor(type), side = tileColorDark(type);
    return '<svg viewBox="0 0 64 48">' +
      '<polygon class="top" points="32,0 64,16 32,32 0,16" fill="'+top+'" stroke="rgba(0,0,0,0.12)" stroke-width="1"/>' +
      (type !== 0 ? '<polygon points="0,16 32,32 32,42 0,26" fill="'+side+'"/><polygon points="64,16 32,32 32,42 64,26" fill="'+side+'"/>' : '') +
      '</svg>';
  }

  var tileEls = {};
  for (var y = 0; y < GRID; y++){
    for (var x = 0; x < GRID; x++){
      var type = map[y][x];
      var el = document.createElement('div');
      el.className = 'tile' + (type > 0 ? ' walkable' : '');
      el.style.left = isoX(x,y) + 'px';
      el.style.top = isoY(x,y) + 'px';
      el.style.zIndex = x + y;
      el.innerHTML = tileSvg(type);
      el.dataset.x = x; el.dataset.y = y; el.dataset.type = type;
      if (type > 0) {
        el.addEventListener('click', function(){ moveTo(parseInt(this.dataset.x), parseInt(this.dataset.y)); });
        el.addEventListener('mouseenter', function(){ this.classList.add('hover'); });
        el.addEventListener('mouseleave', function(){ this.classList.remove('hover'); });
      }
      world.appendChild(el);
      tileEls[x+'_'+y] = el;
    }
  }

  // decorations: palm trees + rocks on sand/grass edges, seeded deterministically
  var seed = 7;
  function rand(){ seed = (seed*9301+49297)%233280; return seed/233280; }
  for (var y2 = 0; y2 < GRID; y2++){
    for (var x2 = 0; x2 < GRID; x2++){
      var t = map[y2][x2];
      if (t === 2 && rand() < 0.1){
        addDeco(x2, y2, 'tree');
      } else if (t === 2 && rand() < 0.08){
        addDeco(x2, y2, 'flower');
      } else if (t === 1 && rand() < 0.08){
        addDeco(x2, y2, 'rock');
      } else if (t === 1 && rand() < 0.05){
        addDeco(x2, y2, 'driftwood');
      }
    }
  }

  function addDeco(x, y, kind){
    var el = document.createElement('div');
    el.className = 'deco';
    el.style.left = isoX(x,y) + 'px';
    el.style.top = (isoY(x,y) + 8) + 'px';
    el.style.zIndex = x + y + 0.5;
    if (kind === 'tree'){
      el.style.width = '54px'; el.style.height = '80px';
      el.innerHTML = '<svg viewBox="0 0 54 80"><rect x="24" y="40" width="6" height="34" fill="#8a5a2b"/><ellipse cx="27" cy="30" rx="22" ry="16" fill="#3f8a3a"/><ellipse cx="15" cy="24" rx="14" ry="11" fill="#4aa044"/><ellipse cx="39" cy="24" rx="14" ry="11" fill="#4aa044"/></svg>';
    } else if (kind === 'rock') {
      el.style.width = '28px'; el.style.height = '20px';
      el.innerHTML = '<svg viewBox="0 0 28 20"><ellipse cx="14" cy="12" rx="13" ry="8" fill="#9a9a92"/><ellipse cx="10" cy="9" rx="6" ry="4" fill="#b0b0a8"/></svg>';
    } else if (kind === 'flower') {
      el.style.width = '20px'; el.style.height = '20px';
      var petal = ['#e8628a','#f2c94c','#8a5ad4'][Math.floor(rand()*3)];
      el.innerHTML = '<svg viewBox="0 0 20 20"><circle cx="10" cy="6" r="3" fill="'+petal+'"/><circle cx="6" cy="9" r="3" fill="'+petal+'"/><circle cx="14" cy="9" r="3" fill="'+petal+'"/><circle cx="10" cy="11" r="3" fill="#f2c94c"/></svg>';
    } else {
      el.style.width = '30px'; el.style.height = '14px';
      el.innerHTML = '<svg viewBox="0 0 30 14"><rect x="0" y="4" width="30" height="5" rx="2" fill="#a9884f"/><rect x="4" y="0" width="22" height="4" rx="2" fill="#c9a86a"/></svg>';
    }
    world.appendChild(el);
  }

  // character — pulled from the welcome screen customizer when available
  var saved = null;
  try { saved = JSON.parse(localStorage.getItem('islandCharacter') || 'null'); } catch(e){}
  var pc = saved || { name: 'Player', gender: 'male', skin: '#f2c9a0', hair: '#7a4a2a', hairStyle: 'a', outfit: '#3d5a8a', cap: '#d94b3f', capOn: true };

  function charHair(gender, style, color){
    if (gender === 'female') {
      if (style === 'a') {
        return '<path d="M4 42 Q-6 66 8 74 Q0 46 12 22 Z" fill="'+color+'"/><path d="M42 42 Q52 66 38 74 Q46 46 34 22 Z" fill="'+color+'"/>' +
               '<path d="M4 40 Q4 4 23 0 Q42 4 42 40 Q34 22 23 22 Q12 22 4 40 Z" fill="'+color+'"/>' +
               '<circle cx="10" cy="14" r="4" fill="#fff2d6" stroke="#e39a3c" stroke-width="1"/>';
      }
      return '<path d="M23 0 Q40 2 42 22 Q46 34 40 44 Q44 30 36 18 Q26 8 23 10 Q20 8 10 18 Q2 30 6 44 Q0 34 4 22 Q6 2 23 0 Z" fill="'+color+'"/>' +
             '<path d="M44 12 Q56 20 52 40 Q50 52 42 56 Q48 42 46 26 Q46 18 44 12 Z" fill="'+color+'"/>' +
             '<circle cx="46" cy="14" r="4" fill="#e8628a"/>';
    }
    if (style === 'a') return '<path d="M4 40 Q4 4 23 0 Q42 4 42 40 Q34 22 23 22 Q12 22 4 40 Z" fill="'+color+'"/>';
    return '<path d="M6 38 L10 14 L15 36 L21 8 L27 36 L33 6 L39 36 L44 12 L40 40 Q23 20 6 38 Z" fill="'+color+'"/>';
  }

  var char = document.createElement('div');
  char.className = 'char';
  char.innerHTML = '<svg viewBox="0 0 46 60">' +
    '<ellipse cx="23" cy="56" rx="12" ry="4" fill="rgba(0,0,0,0.2)"/>' +
    '<rect x="14" y="30" width="18" height="20" rx="7" fill="'+pc.outfit+'"/>' +
    '<rect x="8" y="32" width="8" height="16" rx="4" fill="'+pc.skin+'"/>' +
    '<rect x="30" y="32" width="8" height="16" rx="4" fill="'+pc.skin+'"/>' +
    '<circle cx="23" cy="16" r="14" fill="'+pc.skin+'"/>' +
    charHair(pc.gender, pc.hairStyle, pc.hair) +
    '<circle cx="18" cy="18" r="2" fill="#3a2a1a"/><circle cx="28" cy="18" r="2" fill="#3a2a1a"/>' +
    (pc.capOn ? '<path d="M2 20 Q23 -8 44 20 Q44 10 23 4 Q2 10 2 20 Z" fill="'+pc.cap+'"/><rect x="23" y="2" width="20" height="8" rx="4" fill="'+pc.cap+'"/>' : '') +
    '</svg>';
  var startX = Math.floor(cx), startY = Math.floor(cy);
  var pos = {x: startX, y: startY};
  char.style.left = isoX(pos.x,pos.y) + 'px';
  char.style.top = isoY(pos.x,pos.y) + 'px';
  char.style.zIndex = pos.x + pos.y + 1;
  world.appendChild(char);

  var moving = false;
  function moveTo(tx, ty){
    if (moving) return;
    if (map[ty][tx] === 0) { showToast("Can't walk on water"); return; }
    var path = [];
    var x = pos.x, y = pos.y;
    var steps = Math.max(Math.abs(tx-x), Math.abs(ty-y));
    for (var i = 1; i <= steps; i++){
      var nx = Math.round(x + (tx-x)*i/steps);
      var ny = Math.round(y + (ty-y)*i/steps);
      if (map[ny] && map[ny][nx] > 0) path.push({x:nx,y:ny});
    }
    if (!path.length) return;
    moving = true;
    char.classList.add('walking');
    var idx = 0;
    function step(){
      var p = path[idx];
      char.style.left = isoX(p.x,p.y) + 'px';
      char.style.top = isoY(p.x,p.y) + 'px';
      char.style.zIndex = p.x + p.y + 1;
      pos = p;
      idx++;
      if (idx < path.length) setTimeout(step, 420);
      else { moving = false; char.classList.remove('walking'); }
    }
    step();
  }

  // zoom + pan
  var scale = 1, panX = 0, panY = 0;
  function applyTransform(){
    world.style.transform = 'translate(' + panX + 'px,' + panY + 'px) scale(' + scale + ')';
  }
  applyTransform();

  function setZoom(newScale, cxScreen, cyScreen){
    newScale = Math.max(0.55, Math.min(2, newScale));
    var rect = viewport.getBoundingClientRect();
    var localX = (cxScreen - rect.left - panX) / scale;
    var localY = (cyScreen - rect.top - panY) / scale;
    panX -= localX * (newScale - scale);
    panY -= localY * (newScale - scale);
    scale = newScale;
    applyTransform();
  }

  viewport.addEventListener('wheel', function(e){
    e.preventDefault();
    var factor = e.deltaY < 0 ? 1.1 : 0.9;
    setZoom(scale * factor, e.clientX, e.clientY);
  }, {passive:false});

  document.getElementById('zoomIn').addEventListener('click', function(){
    var r = viewport.getBoundingClientRect();
    setZoom(scale * 1.15, r.left + r.width/2, r.top + r.height/2);
  });
  document.getElementById('zoomOut').addEventListener('click', function(){
    var r = viewport.getBoundingClientRect();
    setZoom(scale * 0.87, r.left + r.width/2, r.top + r.height/2);
  });

  var dragging = false, dragStart = null, panStart = null;
  viewport.addEventListener('mousedown', function(e){
    dragging = false;
    dragStart = {x: e.clientX, y: e.clientY};
    panStart = {x: panX, y: panY};
    viewport.classList.add('dragging');
  });
  window.addEventListener('mousemove', function(e){
    if (!dragStart) return;
    var dx = e.clientX - dragStart.x, dy = e.clientY - dragStart.y;
    if (Math.abs(dx) > 3 || Math.abs(dy) > 3) dragging = true;
    if (dragging){
      panX = panStart.x + dx;
      panY = panStart.y + dy;
      applyTransform();
    }
  });
  window.addEventListener('mouseup', function(){
    dragStart = null;
    viewport.classList.remove('dragging');
    setTimeout(function(){ dragging = false; }, 0);
  });
  viewport.addEventListener('click', function(e){
    if (dragging) e.stopPropagation();
  }, true);

  function showToast(msg){
    var t = document.getElementById('toast');
    t.textContent = msg;
    t.classList.add('show');
    setTimeout(function(){ t.classList.remove('show'); }, 1300);
  }

  if (saved) setTimeout(function(){ showToast('Welcome back, ' + pc.name + '!'); }, 400);
})();
</script>
