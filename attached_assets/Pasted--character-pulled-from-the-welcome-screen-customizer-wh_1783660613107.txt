// movement: walk tile-by-tile from current position to the clicked tile
var moving = false;
function moveTo(tx, ty){
  if (moving) return;                              // ignore clicks mid-walk
  if (map[ty][tx] === 0) { showToast("Can't walk on water"); return; }

  // build a straight-line path of tiles from pos -> (tx,ty)
  var path = [];
  var x = pos.x, y = pos.y;
  var steps = Math.max(Math.abs(tx-x), Math.abs(ty-y));
  for (var i = 1; i <= steps; i++){
    var nx = Math.round(x + (tx-x)*i/steps);
    var ny = Math.round(y + (ty-y)*i/steps);
    if (map[ny] && map[ny][nx] > 0) path.push({x:nx,y:ny}); // only land tiles
  }
  if (!path.length) return;

  moving = true;
  char.classList.add('walking'); // triggers the CSS "bob" animation

  var idx = 0;
  function step(){
    var p = path[idx];
    char.style.left = isoX(p.x,p.y) + 'px';   // CSS transition animates the slide
    char.style.top = isoY(p.x,p.y) + 'px';
    char.style.zIndex = p.x + p.y + 1;
    pos = p;
    idx++;
    if (idx < path.length) setTimeout(step, 420);  // ms per tile — the walk speed
    else { moving = false; char.classList.remove('walking'); }
  }
  step();
}
